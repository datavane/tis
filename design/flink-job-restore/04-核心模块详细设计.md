# 核心模块详细设计

## 4.1 FlinkJobRestoreManager

### 4.1.1 类定义

```java
package com.qlangtech.tis.coredefine.module.action.restore;

/**
 * Flink Job 恢复管理器
 * 职责：
 * 1. TIS Console 启动时从持久化存储加载正在跟踪的 Job 列表
 * 2. 定时轮询 Flink 集群，检测 Job 状态变化
 * 3. 发现 Job 失联/失败后，按策略自动恢复
 * 4. 提供手动 Mapping 接口，绑定已有 Job
 */
@Component
public class FlinkJobRestoreManager implements InitializingBean, DisposableBean {

    private static final Logger logger = LoggerFactory.getLogger(FlinkJobRestoreManager.class);

    // 内存中的监控队列：dataxName -> JobTrackingInfo
    private final ConcurrentHashMap<String, JobTrackingInfo> watchingJobs = 
        new ConcurrentHashMap<>();

    @Autowired
    private FlinkJobStatusStore jobStatusStore;

    @Autowired
    private RestoreStrategy restoreStrategy;

    @Autowired
    private CheckpointCollector checkpointCollector;

    @Autowired
    private JobIdMapper jobIdMapper;

    @Autowired
    private FlinkTaskNodeController taskNodeController;

    // 调度线程池
    private ScheduledExecutorService scheduler;

    @Override
    public void afterPropertiesSet() throws Exception {
        // Console 启动时加载需要跟踪的 Job
        loadTrackingJobs();
        
        // 启动定时监控
        int pollInterval = TisAppLaunch.getFlinkWatcherPollInterval();
        scheduler = Executors.newSingleThreadScheduledExecutor(
            r -> new Thread(r, "flink-job-restore-watcher"));
        scheduler.scheduleWithFixedDelay(
            this::watchAll, 
            pollInterval, 
            pollInterval, 
            TimeUnit.SECONDS
        );
        
        logger.info("FlinkJobRestoreManager started, watching {} jobs", watchingJobs.size());
    }

    @Override
    public void destroy() throws Exception {
        if (scheduler != null) {
            scheduler.shutdownNow();
        }
        // 持久化所有内存状态
        persistAll();
        logger.info("FlinkJobRestoreManager stopped");
    }

    /**
     * 从持久化存储加载正在跟踪的 Job
     */
    private void loadTrackingJobs() {
        List<JobStatus> trackingJobs = jobStatusStore.findByTracking(true);
        for (JobStatus job : trackingJobs) {
            watchingJobs.put(job.getDataxName(), new JobTrackingInfo(job));
        }
    }

    /**
     * 轮询所有正在监控的 Job
     */
    private void watchAll() {
        watchingJobs.values().forEach(this::watchSingle);
    }

    /**
     * 轮询单个 Job 的状态
     */
    private void watchSingle(JobTrackingInfo trackingInfo) {
        String dataxName = trackingInfo.getDataxName();
        String oldJobId = trackingInfo.getJobId();
        
        try {
            // 1. 查询 Flink 集群状态
            JobStatusResult statusResult = queryFlinkJobStatus(dataxName, oldJobId);
            
            // 2. 根据状态结果处理
            handleStatusResult(trackingInfo, statusResult);
            
        } catch (Exception e) {
            logger.error("Failed to watch job: {} (jobId: {})", dataxName, oldJobId, e);
            trackingInfo.incrementLostCount();
            
            // 连续多次查询失败，判定为失联
            if (trackingInfo.getLostCount() >= getLostThreshold()) {
                handleLost(trackingInfo);
            }
        }
    }

    /**
     * 查询 Flink Job 状态
     */
    private JobStatusResult queryFlinkJobStatus(String dataxName, String jobId) {
        // 复用 FlinkTaskNodeController 的查询逻辑
        IFlinkIncrJobStatus jobStatus = FlinkIncrJobStatus.load(dataxName);
        
        try {
            // 尝试通过旧 JobID 查询
            FlinkJobDeploymentDetails details = 
                taskNodeController.getRCDeployment(new TargetResName(dataxName));
            
            if (details != null && details.isRunning()) {
                return JobStatusResult.running(details.getJobId());
            }
        } catch (JobNotFoundException e) {
            // JobID 在集群中不存在，尝试通过 JobName 匹配
            String matchedJobId = jobIdMapper.findByJobName(dataxName);
            if (matchedJobId != null) {
                return JobStatusResult.running(matchedJobId, "matched_by_name");
            }
        }
        
        return JobStatusResult.notFound();
    }

    /**
     * 根据状态结果处理
     */
    private void handleStatusResult(JobTrackingInfo trackingInfo, JobStatusResult result) {
        switch (result.getState()) {
            case RUNNING:
                handleRunning(trackingInfo, result);
                break;
            case NOT_FOUND:
                handleNotFound(trackingInfo);
                break;
            case FAILED:
                handleFailed(trackingInfo);
                break;
            default:
                logger.warn("Unknown job state: {} for {}", result.getState(), 
                           trackingInfo.getDataxName());
        }
    }

    /**
     * Job 正常运行
     */
    private void handleRunning(JobTrackingInfo trackingInfo, JobStatusResult result) {
        // 如果 JobID 发生了变化（如 HA 恢复后新 ID），更新跟踪信息
        if (!StringUtils.equals(trackingInfo.getJobId(), result.getJobId())) {
            logger.info("JobID changed for {}: {} -> {}, reason: {}",
                       trackingInfo.getDataxName(), 
                       trackingInfo.getJobId(), 
                       result.getJobId(),
                       result.getChangeReason());
            trackingInfo.setJobId(result.getJobId());
            jobStatusStore.updateJobId(trackingInfo.getDataxName(), result.getJobId());
        }
        
        trackingInfo.resetLostCount();
        trackingInfo.updateLastSeenTime();
        trackingInfo.setState(JobState.RUNNING);
    }

    /**
     * Job 在集群中不存在（可能集群已重启）
     */
    private void handleNotFound(JobTrackingInfo trackingInfo) {
        trackingInfo.incrementLostCount();
        logger.warn("Job not found: {} (jobId: {}), lostCount: {}",
                   trackingInfo.getDataxName(), 
                   trackingInfo.getJobId(),
                   trackingInfo.getLostCount());
        
        if (trackingInfo.getLostCount() >= getLostThreshold()) {
            handleLost(trackingInfo);
        }
    }

    /**
     * Job 失联，触发自动恢复
     */
    private void handleLost(JobTrackingInfo trackingInfo) {
        String dataxName = trackingInfo.getDataxName();
        logger.warn("Job {} is LOST, triggering restore process", dataxName);
        
        trackingInfo.setState(JobState.LOST);
        jobStatusStore.updateState(dataxName, JobState.LOST);
        
        // 尝试自动恢复
        if (restoreStrategy.shouldAutoRestore(trackingInfo)) {
            executeRestore(trackingInfo);
        } else {
            logger.warn("Auto restore disabled or exceeded limit for {}", dataxName);
            trackingInfo.setState(JobState.TERMINATED);
            jobStatusStore.updateState(dataxName, JobState.TERMINATED);
            watchingJobs.remove(dataxName);
        }
    }

    /**
     * Job 失败
     */
    private void handleFailed(JobTrackingInfo trackingInfo) {
        logger.error("Job {} is FAILED, triggering restore process", 
                    trackingInfo.getDataxName());
        trackingInfo.setState(JobState.FAILED);
        jobStatusStore.updateState(trackingInfo.getDataxName(), JobState.FAILED);
        executeRestore(trackingInfo);
    }

    /**
     * 执行恢复操作
     */
    private void executeRestore(JobTrackingInfo trackingInfo) {
        String dataxName = trackingInfo.getDataxName();
        
        try {
            trackingInfo.setState(JobState.RESTARTING);
            jobStatusStore.updateState(dataxName, JobState.RESTARTING);
            
            // 1. 获取恢复点
            RestorePoint restorePoint = checkpointCollector.getLatestRestorePoint(dataxName);
            if (restorePoint == null) {
                throw new RestoreException("No available restore point found for " + dataxName);
            }
            
            logger.info("Restoring job {} from {}", dataxName, restorePoint.getPath());
            
            // 2. 重新提交 Job
            JobID newJobId = taskNodeController.relaunch(
                new TargetResName(dataxName),
                restorePoint.getPath(),
                restorePoint.getRestoreMode()
            );
            
            // 3. 更新状态
            trackingInfo.setJobId(newJobId.toHexString());
            trackingInfo.setState(JobState.RUNNING);
            trackingInfo.setRestartCount(trackingInfo.getRestartCount() + 1);
            trackingInfo.resetLostCount();
            
            jobStatusStore.updateAfterRestore(dataxName, newJobId.toHexString(), 
                                              JobState.RUNNING, 
                                              trackingInfo.getRestartCount());
            
            // 4. 记录恢复历史
            recordRestoreHistory(dataxName, trackingInfo.getOldJobId(), 
                                newJobId.toHexString(), restorePoint, true, null);
            
            logger.info("Job {} restored successfully, new jobId: {}", 
                       dataxName, newJobId.toHexString());
            
        } catch (Exception e) {
            logger.error("Failed to restore job {}", dataxName, e);
            trackingInfo.setState(JobState.RESTORE_FAILED);
            jobStatusStore.updateState(dataxName, JobState.RESTORE_FAILED);
            
            recordRestoreHistory(dataxName, trackingInfo.getJobId(), 
                                null, null, false, e.getMessage());
        }
    }

    /**
     * 手动认领一个运行中的 Job
     */
    public void manualMapJob(String dataxName, String newJobId, String clusterAddress) {
        JobTrackingInfo trackingInfo = watchingJobs.get(dataxName);
        if (trackingInfo == null) {
            // 新建跟踪记录
            trackingInfo = new JobTrackingInfo();
            trackingInfo.setDataxName(dataxName);
            watchingJobs.put(dataxName, trackingInfo);
        }
        
        trackingInfo.setJobId(newJobId);
        trackingInfo.setClusterAddress(clusterAddress);
        trackingInfo.setState(JobState.RUNNING);
        trackingInfo.setTracking(true);
        trackingInfo.resetLostCount();
        
        jobStatusStore.updateAfterManualMap(dataxName, newJobId, clusterAddress);
        
        logger.info("Job {} manually mapped to new jobId: {}", dataxName, newJobId);
    }

    /**
     * 持久化所有内存状态
     */
    private void persistAll() {
        watchingJobs.values().forEach(info -> 
            jobStatusStore.save(JobStatus.fromTrackingInfo(info))
        );
    }

    private int getLostThreshold() {
        return TisAppLaunch.getFlinkLostThreshold();
    }
}
```

### 4.1.2 JobTrackingInfo（内存跟踪对象）

```java
@Data
public class JobTrackingInfo {
    private String dataxName;           // DataX Pipeline 名称
    private String jobId;               // 当前 Flink JobID
    private String oldJobId;            // 恢复前的旧 JobID
    private String clusterAddress;      // Flink 集群地址
    private JobState state;             // 当前状态
    private boolean tracking;           // 是否正在跟踪
    private int restartCount;           // 已自动重启次数
    private int lostCount;              // 连续失联计数
    private long lastSeenTime;          // 最后一次看到 Job 的时间
    private long lastRestoreTime;       // 最后一次恢复时间

    public JobTrackingInfo(JobStatus jobStatus) {
        this.dataxName = jobStatus.getDataxName();
        this.jobId = jobStatus.getJobId();
        this.clusterAddress = jobStatus.getClusterAddress();
        this.state = jobStatus.getState();
        this.tracking = jobStatus.isTracking();
        this.restartCount = jobStatus.getRestartCount();
        this.lastSeenTime = System.currentTimeMillis();
    }

    public void incrementLostCount() {
        this.lostCount++;
    }

    public void resetLostCount() {
        this.lostCount = 0;
    }

    public void updateLastSeenTime() {
        this.lastSeenTime = System.currentTimeMillis();
    }
}
```

### 4.2.2 默认实现：AutoRestartStrategy

```java
@Component
public class AutoRestartStrategy implements RestoreStrategy {

    private static final Logger logger = 
        LoggerFactory.getLogger(AutoRestartStrategy.class);

    @Override
    public boolean shouldAutoRestore(JobTrackingInfo trackingInfo) {
        // 1. 检查全局开关
        if (!TisAppLaunch.isFlinkJobRestoreEnabled()) {
            logger.debug("Auto restore is globally disabled");
            return false;
        }
        
        // 2. 检查任务级开关
        if (!trackingInfo.isAutoRestoreEnabled()) {
            logger.debug("Auto restore is disabled for job: {}", 
                        trackingInfo.getDataxName());
            return false;
        }
        
        // 3. 检查最大重启次数
        int maxRestarts = TisAppLaunch.getFlinkRestoreMaxRestarts();
        if (trackingInfo.getRestartCount() >= maxRestarts) {
            logger.warn("Auto restore exceeded max restarts ({}) for job: {}",
                       maxRestarts, trackingInfo.getDataxName());
            return false;
        }
        
        // 4. 检查恢复间隔
        long minIntervalMs = TisAppLaunch.getFlinkRestoreMinIntervalSec() * 1000L;
        if (trackingInfo.getLastRestoreTime() > 0 
            && System.currentTimeMillis() - trackingInfo.getLastRestoreTime() < minIntervalMs) {
            logger.debug("Restore interval too short for job: {}", 
                        trackingInfo.getDataxName());
            return false;
        }
        
        // 5. 检查是否有可用的恢复点
        //（在 executeRestore 中再详细检查）
        
        return true;
    }

    @Override
    public RestorePoint selectRestorePoint(String dataxName, 
                                            List<RestorePoint> candidates) {
        if (candidates == null || candidates.isEmpty()) {
            return null;
        }
        
        // 优先选择 Savepoint（更稳定）
        Optional<RestorePoint> savepoint = candidates.stream()
            .filter(cp -> cp.getType() == RestorePointType.SAVEPOINT)
            .max(Comparator.comparing(RestorePoint::getTriggerTime));
        
        if (savepoint.isPresent()) {
            return savepoint.get();
        }
        
        // 次选最新的 Checkpoint
        return candidates.stream()
            .max(Comparator.comparing(RestorePoint::getTriggerTime))
            .orElse(null);
    }

    @Override
    public FlinkRestoreMode getRestoreMode() {
        return TisAppLaunch.getFlinkRestoreMode();
    }
}
```

```java
package com.qlangtech.tis.coredefine.module.action.restore;

/**
 * JobID 映射器
 * 用于在 JobID 变化后，尝试找到对应的新 Job
 */
@Component
public class JobIdMapper {

    private static final Logger logger = LoggerFactory.getLogger(JobIdMapper.class);

    @Autowired
    private FlinkCluster flinkCluster;

    /**
     * 通过 JobName 在 Flink 集群中查找对应的 JobID
     * 适用场景：Flink HA 恢复后 JobID 变化，但 JobName 不变
     */
    public String findByJobName(String dataxName) {
        try {
            // 获取 Flink 集群上所有运行的 Job
            RestClusterClient<StandaloneClusterId> client = 
                flinkCluster.createRestClusterClient();
            
            Collection<JobStatusMessage> jobs = client.listJobs().get();
            
            // TIS 提交的 Job 名称通常与 dataxName 相关
            String expectedJobName = generateExpectedJobName(dataxName);
            
            for (JobStatusMessage job : jobs) {
                if (StringUtils.contains(job.getJobName(), expectedJobName)) {
                    logger.info("Found matching job by name: {} -> jobId: {}",
                               dataxName, job.getJobId().toHexString());
                    return job.getJobId().toHexString();
                }
            }
            
        } catch (Exception e) {
            logger.error("Failed to find job by name: {}", dataxName, e);
        }
        
        return null;
    }

    /**
     * 生成预期的 JobName
     * TIS 提交的 Flink Job 名称规则需要与提交时保持一致
     */
    private String generateExpectedJobName(String dataxName) {
        // 例如：TIS 提交时使用的 JobName 格式
        return "TIS-" + dataxName;
    }

    /**
     * 通过 Savepoint 路径反查 JobID
     * 适用场景：知道 Savepoint 属于哪个 Job
     */
    public String findBySavepointPath(String savepointPath) {
        // Flink REST API 目前没有直接通过 Savepoint 反查 JobID 的接口
        // 此方法的实现需要依赖外部元数据
        return null;
    }
}
```

## 4.4 CheckpointCollector（恢复点收集器）

### 4.4.1 类定义

```java
package com.qlangtech.tis.coredefine.module.action.restore;

/**
 * Checkpoint/Savepoint 收集器
 * 负责定时收集 Flink Job 的恢复点信息
 */
@Component
public class CheckpointCollector {

    private static final Logger logger = 
        LoggerFactory.getLogger(CheckpointCollector.class);

    @Autowired
    private FlinkJobStatusStore jobStatusStore;

    @Autowired
    private FlinkTaskNodeController taskNodeController;

    private ScheduledExecutorService scheduler;

    @PostConstruct
    public void init() {
        if (!TisAppLaunch.isCheckpointCollectEnabled()) {
            logger.info("Checkpoint collector is disabled");
            return;
        }
        
        int interval = TisAppLaunch.getCheckpointCollectIntervalSec();
        scheduler = Executors.newSingleThreadScheduledExecutor(
            r -> new Thread(r, "checkpoint-collector"));
        scheduler.scheduleWithFixedDelay(
            this::collectAll, interval, interval, TimeUnit.SECONDS
        );
        
        logger.info("CheckpointCollector started, interval: {}s", interval);
    }

    @PreDestroy
    public void destroy() {
        if (scheduler != null) {
            scheduler.shutdownNow();
        }
    }

    /**
     * 收集所有正在跟踪的 Job 的 Checkpoint 信息
     */
    private void collectAll() {
        List<JobStatus> trackingJobs = jobStatusStore.findByTracking(true);
        
        for (JobStatus job : trackingJobs) {
            try {
                collectSingle(job);
            } catch (Exception e) {
                logger.error("Failed to collect checkpoints for job: {}", 
                            job.getDataxName(), e);
            }
        }
    }

    /**
     * 收集单个 Job 的 Checkpoint 信息
     */
    private void collectSingle(JobStatus job) {
        String dataxName = job.getDataxName();
        String jobId = job.getJobId();
        
        if (StringUtils.isBlank(jobId)) {
            return;
        }
        
        // 通过 Flink REST API 获取 Checkpoint 信息
        // 复用 CoreAction.doGetJobCheckpoints 的代理逻辑
        JobCheckpoints checkpoints = fetchCheckpointsFromFlink(jobId, job);
        
        if (checkpoints == null || checkpoints.getHistory() == null) {
            return;
        }
        
        // 解析 completed checkpoints
        for (JobCheckpointInfo checkpoint : checkpoints.getHistory()) {
            if ("COMPLETED".equals(checkpoint.getStatus())) {
                RestorePoint restorePoint = RestorePoint.builder()
                    .dataxName(dataxName)
                    .checkpointId(checkpoint.getId())
                    .path(checkpoint.getExternalPath())
                    .type(RestorePointType.CHECKPOINT)
                    .triggerTime(checkpoint.getTriggerTimestamp())
                    .latest(false)  // 先标记为非最新，后续统一更新
                    .build();
                
                jobStatusStore.saveRestorePoint(restorePoint);
            }
        }
        
        // 标记最新的恢复点
        jobStatusStore.markLatestRestorePoint(dataxName);
        
        logger.debug("Collected {} checkpoints for job: {}", 
                    checkpoints.getHistory().size(), dataxName);
    }

    /**
     * 从 Flink 集群获取 Checkpoint 信息
     */
    private JobCheckpoints fetchCheckpointsFromFlink(String jobId, JobStatus job) {
        try {
            String jmUrl = job.getClusterAddress();
            if (StringUtils.isBlank(jmUrl)) {
                // 尝试从配置获取
                jmUrl = getJobManagerUrl(job.getDataxName());
            }
            
            String checkpointsUrl = jmUrl + "/jobs/" + jobId + "/checkpoints";
            // 使用 HTTP Client 调用 Flink REST API
            String response = HttpUtils.get(checkpointsUrl);
            return JSON.parseObject(response, JobCheckpoints.class);
            
        } catch (Exception e) {
            logger.error("Failed to fetch checkpoints from Flink for job: {}", jobId, e);
            return null;
        }
    }

    /**
     * 获取最新的可用恢复点
     */
    public RestorePoint getLatestRestorePoint(String dataxName) {
        // 1. 优先从数据库查询
        RestorePoint latest = jobStatusStore.findLatestRestorePoint(dataxName);
        if (latest != null && verifyRestorePointExists(latest)) {
            return latest;
        }
        
        // 2. 尝试从本地文件读取
        IFlinkIncrJobStatus fileStatus = FlinkIncrJobStatus.load(dataxName);
        if (fileStatus != null) {
            List<SavePointMsg> savepoints = fileStatus.getSavePointPaths();
            if (savepoints != null && !savepoints.isEmpty()) {
                SavePointMsg sp = savepoints.get(savepoints.size() - 1);
                return RestorePoint.builder()
                    .dataxName(dataxName)
                    .path(sp.getPath())
                    .type(RestorePointType.SAVEPOINT)
                    .triggerTime(sp.getCreateTime())
                    .build();
            }
        }
        
        return null;
    }

    /**
     * 验证恢复点路径是否存在
     */
    private boolean verifyRestorePointExists(RestorePoint restorePoint) {
        try {
            Path path = new Path(restorePoint.getPath());
            FileSystem fs = FileSystem.get(path.toUri(), new Configuration());
            return fs.exists(path);
        } catch (Exception e) {
            logger.warn("Failed to verify restore point path: {}", 
                       restorePoint.getPath(), e);
            return false;
        }
    }
}
```

### 4.5.2 数据库实现

```java
@Primary
@Component
public class DatabaseJobStatusStore implements FlinkJobStatusStore {

    @Autowired
    private FlinkJobStatusMapper jobStatusMapper;

    @Autowired
    private FlinkRestorePointMapper restorePointMapper;

    @Override
    public void save(JobStatus jobStatus) {
        FlinkJobStatusRecord record = convertToRecord(jobStatus);
        
        FlinkJobStatusRecord existing = jobStatusMapper.selectByDataxName(
            jobStatus.getDataxName()
        );
        
        if (existing == null) {
            jobStatusMapper.insert(record);
        } else {
            record.setId(existing.getId());
            jobStatusMapper.updateById(record);
        }
    }

    @Override
    public JobStatus findByDataxName(String dataxName) {
        FlinkJobStatusRecord record = jobStatusMapper.selectByDataxName(dataxName);
        return record != null ? convertFromRecord(record) : null;
    }

    @Override
    public List<JobStatus> findByTracking(boolean tracking) {
        List<FlinkJobStatusRecord> records = jobStatusMapper.selectList(
            new QueryWrapper<FlinkJobStatusRecord>()
                .eq("tracking", tracking ? 1 : 0)
        );
        return records.stream()
            .map(this::convertFromRecord)
            .collect(Collectors.toList());
    }

    @Override
    public void updateState(String dataxName, JobState state) {
        jobStatusMapper.update(null, 
            new UpdateWrapper<FlinkJobStatusRecord>()
                .set("state", state.name())
                .set("update_time", new Date())
                .eq("datax_name", dataxName)
        );
    }

    @Override
    public void updateJobId(String dataxName, String jobId) {
        jobStatusMapper.update(null,
            new UpdateWrapper<FlinkJobStatusRecord>()
                .set("job_id", jobId)
                .set("update_time", new Date())
                .eq("datax_name", dataxName)
        );
    }

    @Override
    public void updateAfterRestore(String dataxName, String newJobId, 
                                    JobState state, int restartCount) {
        jobStatusMapper.update(null,
            new UpdateWrapper<FlinkJobStatusRecord>()
                .set("job_id", newJobId)
                .set("state", state.name())
                .set("restart_count", restartCount)
                .set("last_restore_time", new Date())
                .set("update_time", new Date())
                .eq("datax_name", dataxName)
        );
    }

    @Override
    public void saveRestorePoint(RestorePoint restorePoint) {
        FlinkRestorePointRecord record = convertToRecord(restorePoint);
        restorePointMapper.insert(record);
        
        // 清理旧的历史记录，保留最近 N 条
        cleanupOldRestorePoints(restorePoint.getDataxName());
    }

    @Override
    public RestorePoint findLatestRestorePoint(String dataxName) {
        FlinkRestorePointRecord record = restorePointMapper.selectLatest(dataxName);
        return record != null ? convertFromRecord(record) : null;
    }

    @Override
    public void markLatestRestorePoint(String dataxName) {
        // 1. 清除旧的 latest 标记
        restorePointMapper.update(null,
            new UpdateWrapper<FlinkRestorePointRecord>()
                .set("latest", 0)
                .eq("datax_name", dataxName)
        );
        
        // 2. 标记最新的为 latest
        restorePointMapper.updateLatest(dataxName);
    }

    private void cleanupOldRestorePoints(String dataxName) {
        int maxHistory = TisAppLaunch.getCheckpointMaxHistory();
        restorePointMapper.deleteOldRecords(dataxName, maxHistory);
    }

    // ... convert methods
}
```

```java
public void doEnableAutoRestore(Context context) {

    boolean enabled = context.getBoolean("enabled", true);

    status.setMaxRestarts(maxRestarts);

    if (enabled && !status.isTracking()) {
        restoreManager.addToWatchQueue(dataxName);
    }
    
    this.setBizResult(context, new JSONObject()
        .fluentPut("success", true)
        .fluentPut("dataxName", dataxName)
        .fluentPut("enabled", enabled)
        .fluentPut("maxRestarts", maxRestarts));
}
```

```java
/**
 * 手动认领一个运行中的 Flink Job
 * 适用场景：自动恢复失败，用户手动找到新 JobID 后绑定
 */
public void doManualMapJob(Context context) {
    String dataxName = context.getString("dataxName");
    String newJobId = context.getString("jobId");
    String clusterAddress = context.getString("clusterAddress");
    
    // 验证 JobID 是否有效

    if (!valid) {
        throw new IllegalArgumentException(
            "Invalid jobId or cluster address: " + newJobId);
    }
    
    // 执行手动映射

    this.setBizResult(context, new JSONObject()
        .fluentPut("success", true)
        .fluentPut("dataxName", dataxName)
        .fluentPut("jobId", newJobId)
        .fluentPut("state", "RUNNING"));
}

private boolean verifyJobId(String dataxName, String jobId, String clusterAddress) {
    try {
        RestClusterClient client = createRestClient(clusterAddress);
        JobStatusMessage status = client.getJobStatus(JobID.fromHexString(jobId)).get();
        return status != null;
    } catch (Exception e) {
        logger.error("Failed to verify jobId: {}", jobId, e);
        return false;
    }
}
```
